  lastFrame = millis();
  screen.setRotation(0); // launcher already initialized the TFT
  screen.fillScreen(ST77XX_BLACK);
  screen.setTextSize(2);
  screen.setTextColor(ST77XX_WHITE);
  screen.setCursor(20, 30);
  screen.print("MY GAME");
}

    lastFrame = millis();

  }
}
} // namespace MyGame
